Histogram  
import matplotlib.pyplot as plt 
import numpy as np 
data = np.random.normal(loc=75, scale=10, size=40)  
plt.figure(figsize=(6, 4)) 
plt.hist(data, bins=5, color='black', edgecolor='black', alpha=0.7) 
plt.title('Histogram of Test Scores') 
plt.xlabel('Test Scores') 
plt.ylabel('Frequency') 
plt.tight_layout() 
plt.show()
