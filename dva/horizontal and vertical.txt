Aim: Write a Python program to create Horizontal and Vertical orientated bar charts. 
Source Code:  
Vertical Bar Plot 
import matplotlib.pyplot as plt 
import numpy as np 
regions = ['North', 'South', 'East', 'West'] 
sales = [200, 220, 250, 190] 
positions = np.arange(len(regions)) 
plt.figure(figsize=(6, 4)) 
plt.bar(positions, sales, color='black') 
plt.xticks(positions, regions) 
plt.xlabel('Region') 
plt.ylabel('Sales') 
plt.title('Vertical Bar Chart of Sales by Region') 
plt.show() 


Horizontal Bar Plot  
import matplotlib.pyplot as plt 
import numpy as np 
import matplotlib.pyplot as plt 
import numpy as np 
regions = ['North', 'South', 'East', 'West'] 
sales = [200, 220, 250, 190] 
positions = np.arange(len(regions)) 
plt.figure(figsize=(6, 4)) 
plt.barh(positions, sales, color='black') 
plt.yticks(positions, regions) 
plt.ylabel('Region') 
plt.xlabel('Sales') 
plt.title('Horizontal Bar Chart of Sales by Region') 
plt.show()
